//   1. Firebase Console -> Firestore Database -> pestaña "Reglas"
//   2. Borra lo que haya y pega TODO lo de abajo
//      (desde "rules_version" hasta la última llave)
//   3. Click "Publicar"
//
// QUÉ HACEN ESTAS REGLAS:
//   - productos: cualquiera puede VER, solo admin puede MODIFICAR
//   - pedidos:   cada usuario solo ve/crea SUS pedidos
//   - Nadie puede borrar pedidos desde la web
// ============================================================

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // PRODUCTOS: lectura pública, escritura solo autenticado
    // (más adelante se puede restringir a un email admin específico)
    match /productos/{id} {
      allow read: if true;
      allow write: if request.auth != null;
    }

    // PEDIDOS: cada usuario solo accede a los suyos
    match /pedidos/{id} {
      allow create: if request.auth != null
                    && request.resource.data.usuarioId == request.auth.uid;
      allow read:   if request.auth != null
                    && resource.data.usuarioId == request.auth.uid;
      allow update, delete: if false; // nadie edita/borra pedidos desde la web
    }
  }
}
